\section*{Glossário}

\begin{itemize}[leftmargin=1.2cm]
  \item \textbf{Auftragstaktik}: doutrina alemã de comando descentralizado, dando autonomia tática a oficiais subalternos.
  \item \textbf{Fallschirmjäger}: paraquedistas de elite da Luftwaffe alemã.
  \item \textbf{G.I.}: apelido para o soldado raso do exército americano (``Government Issue'').
  \item \textbf{HE (High Explosive)}: categoria de munição explosiva usada por morteiros, artilharia e certas armas anti-tanque.
  \item \textbf{Heer}: o exército de terra regular da Alemanha nazista.
  \item \textbf{HMG}: sigla para Metralhadora Pesada (Heavy Machine Gun).
  \item \textbf{LMG}: sigla para Metralhadora Leve (Light Machine Gun).
  \item \textbf{NCO}: sigla para oficial não-comissionado (Non-Commissioned Officer), como um sargento.
  \item \textbf{Panzerschreck}: lança-foguetes anti-tanque alemão, equivalente à Bazooka americana.
  \item \textbf{SMG}: sigla para submetralhadora (Submachine Gun).
  \item \textbf{Waffen-SS}: ramo de combate armado da SS, força de elite ideológica do regime nazista.
  \item \textbf{Wehrmacht}: nome coletivo para as forças armadas alemãs durante o período nazista.
\end{itemize}
